# Win Net Tempo Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A win net that knows who moves next (a symmetric tempo head beside the antisymmetric logit), every evaluation site saying who moves next, and `win-arm --tempo` to fit it.

**Architecture:** The tempo head is one W→1 `Linear` on the trunk's last hidden layer, stored in `tempo.mpk` beside `model.mpk` (as the aux heads are), so `WinModel` and every existing dir are unchanged. A `Next { Opp, Me }` flag joins the `Bot` and `WinProb` traits through new `*_next` methods whose defaults ignore it; `WinStrategy` honours it only when its net has a tempo head, so every existing bot is bit-for-bit unchanged. Passive decisions and search leaves where seat 0 moves next pass `Next::Me`.

**Tech Stack:** Rust (edition 2021), burn 0.20.1 (ndarray backend, autodiff for fitting), clap. Tests: `cargo test --release --lib <path>`.

**Spec:** `docs/superpowers/specs/2026-10-20-win-tempo-design.md`

```rust
/// A tempo head (W → 1, with bias) that adds nothing: a net given it plays
/// exactly as the antisymmetric net it extends.
pub fn zero_tempo<B: Backend>(width: usize, device: &B::Device) -> Linear<B> {
    let mut head = LinearConfig::new(width, 1).with_bias(true).init(device);
    head.weight = Param::from_tensor(Tensor::zeros([width, 1], device));
    head.bias = Some(Param::from_tensor(Tensor::zeros([1], device)));
    head
}
```

```rust
    fn framed_logits(&self, items: &[(&State, State, Next)]) -> Vec<f32> {
        let pairs: Vec<(&State, &State)> = items
            .iter()
            .map(|(c, o, n)| match n {
                Next::Opp => (*c, o),
                Next::Me => (o, *c),
            })
            .collect();
        self.net
            .logit_batch(&pairs)
            .into_iter()
            .zip(items)
            .map(|(z, (_, _, n))| if *n == Next::Me { -z } else { z })
            .collect()
    }
```

```rust
//! Test helper: a bot that logs the frame (`Next`) of every evaluation.

use super::search::WinProb;
use super::{Bot, Next};
use crate::state::State;
use std::cell::RefCell;

/// One logged evaluation: its frame, whether it was a search leaf, and the
/// strikes of the group's first candidate (or of the leaf's own board).
#[derive(Clone, Copy, Debug, PartialEq)]
pub struct Call {
    pub next: Next,
    pub leaf: bool,
    pub strikes: u8,
}

#[derive(Debug, Default)]
pub struct Recorder {
    calls: RefCell<Vec<Call>>,
}

impl Recorder {
    /// The calls since the last `take`.
    pub fn take(&self) -> Vec<Call> {
        self.calls.take()
    }

    fn value(s: &State, opps: &[State]) -> f32 {
        let best = opps.iter().map(|o| o.count_points()).max().unwrap_or(0);
        (s.count_points() - best) as f32 / 10.0
    }

    fn log(&self, next: Next, leaf: bool, s: Option<&State>) {
        if let Some(s) = s {
            self.calls.borrow_mut().push(Call {
                next,
                leaf,
                strikes: s.strikes,
            });
        }
    }
}

impl Bot for Recorder {
    fn evaluate(&self, our_state: &State, opp_states: &[State]) -> f32 {
        self.evaluate_batch(std::slice::from_ref(our_state), opp_states)[0]
    }

    fn evaluate_batch(&self, candidates: &[State], opp_states: &[State]) -> Vec<f32> {
        self.evaluate_batch_next(candidates, opp_states, Next::Opp)
    }

    fn evaluate_batch_multi(&self, groups: &[(&[State], &[State])]) -> Vec<Vec<f32>> {
        groups.iter().map(|(c, o)| self.evaluate_batch_next(c, o, Next::Opp)).collect()
    }

    fn evaluate_batch_next(&self, candidates: &[State], opp_states: &[State], next: Next) -> Vec<f32> {
        self.log(next, false, candidates.first());
        candidates.iter().map(|s| Self::value(s, opp_states)).collect()
    }

    fn evaluate_batch_multi_next(&self, groups: &[(&[State], &[State], Next)]) -> Vec<Vec<f32>> {
        groups.iter().map(|(c, o, n)| self.evaluate_batch_next(c, o, *n)).collect()
    }
}

impl WinProb for Recorder {
    fn win_prob_multi(&self, groups: &[(&State, &[State])]) -> Vec<f32> {
        let framed: Vec<(&State, &[State], Next)> = groups.iter().map(|(s, o)| (*s, *o, Next::Opp)).collect();
        self.win_prob_multi_next(&framed)
    }

    fn win_prob_multi_next(&self, groups: &[(&State, &[State], Next)]) -> Vec<f32> {
        groups
            .iter()
            .map(|(s, o, n)| {
                self.log(*n, true, Some(s));
                1.0 / (1.0 + (-Self::value(s, o)).exp())
            })
            .collect()
    }
}
```

```rust
    /// Mirror pairs, each side "just moved" with label 0.45: the player to
    /// move next has the edge. Only a tempo net can fit both at 0.45.
    fn mirror_data(dir: &str) {
        let mut rng = SmallRng::seed_from_u64(9);
        let mut board = || {
            let mut s = State::default();
            for _ in 0..rng.gen_range(0..12) {
                let (row, number) = (rng.gen_range(0..4), rng.gen_range(2..=12));
                if s.can_mark(row, number) {
                    s.apply_mark(Mark { row, number });
                }
            }
            s
        };
        let sample = |me: State, opp: State| Sample {
            me,
            opp,
            win: 0.45,
            win_var: 0.25,
            marks: [0.0; 8],
            strikes: [0.0; 2],
            diff: 0.0,
        };
        let mut split = || {
            let mut turns = Vec::new();
            for _ in 0..200 {
                let (a, b) = (board(), board());
                turns.push(sample(a, b));
                turns.push(sample(b, a));
            }
            Split {
                turns,
                decisions: Vec::new(),
            }
        };
        let (train, heldout) = (split(), split());
        let meta = win_data::DataMeta {
            player: "synthetic".into(),
            seed: 0,
            games: 2,
            rollouts: 64,
            cand_rollouts: 64,
            holdout: 0.5,
            keep: None,
            train_turns: 0,
            train_decisions: 0,
            heldout_turns: 0,
            heldout_decisions: 0,
            bytes: 0,
        };
        save(dir, &Dataset { meta, train, heldout }).unwrap();
    }

    #[test]
    fn a_tempo_arm_learns_the_value_of_moving_next() {
        let base = tmp("tempo-learns");
        let data = format!("{base}/data");
        mirror_data(&data);
        let ds = load(&data).unwrap();
        let mean_p = |out: &str| {
            let bot = WinStrategy::load(out);
            let leaves: Vec<(&State, &[State])> = ds
                .heldout
                .turns
                .iter()
                .map(|s| (&s.me, std::slice::from_ref(&s.opp)))
                .collect();
            let p = bot.win_prob_multi(&leaves);
            p.iter().sum::<f32>() / p.len() as f32
        };
        let run = |out: String, tempo: bool| {
            fit(&ArmCfg {
                tempo,
                epochs: 40,
                lr: 1e-2,
                ema: 0.0,
                width: 8,
                ..arm(&data, out.clone(), None)
            })
            .unwrap();
            mean_p(&out)
        };
        let plain = run(format!("{base}/plain"), false);
        let tempo = run(format!("{base}/tempo"), true);
        assert!((plain - 0.5).abs() < 1e-3, "antisymmetric over mirror pairs: {plain}");
        assert!(tempo < 0.47, "the tempo net learns the mover's edge: {tempo}");
        std::fs::remove_dir_all(&base).unwrap();
    }
```
